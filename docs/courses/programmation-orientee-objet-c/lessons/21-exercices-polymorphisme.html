public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }
    public Media(string titre, int annee) { Titre = titre; Annee = annee; }
    public virtual string GetInfo() => $"{Titre} ({Annee})";

public class Film : Media

    public override string GetInfo() => $"{Titre} ({Annee}) - {Realisateur}, {DureeMinutes} min";

Exercice 2 : CalculerPrix polymorphe

Ajoutez une méthode virtuelle CalculerPrix() dans Media :

  • Media : prix de base 9.99 EUR
  • Film : 14.99 EUR si > 120 min, sinon 9.99 EUR
  • Serie : 7.99 EUR par saison
  • Documentaire : 6.99 EUR

Créez une méthode AfficherFacture(List<Media> medias) qui calcule le total.

public class Media
{
    // TODO : ajoutez CalculerPrix virtual
}

static void AfficherFacture(List<Media> medias)
{
    // TODO
}
Solution
// Dans Media :
public virtual decimal CalculerPrix() => 9.99m;

// Dans Film :
public override decimal CalculerPrix() => DureeMinutes > 120 ? 14.99m : 9.99m;

// Dans Serie :
public override decimal CalculerPrix() => Saisons * 7.99m;

// Dans Documentaire :
public override decimal CalculerPrix() => 6.99m;

static void AfficherFacture(List<Media> medias)
{
    decimal total = 0;
    foreach (var m in medias)
    {
        decimal prix = m.CalculerPrix();
        Console.WriteLine($"{m.GetInfo()} : {prix:C}");
        total += prix;
    }
    Console.WriteLine(new string('-', 40));
    Console.WriteLine($"TOTAL : {total:C}");
}

Exercice 3 : Redéfinition de ToString()

Overridez ToString() dans chaque classe pour produire un affichage complet et différencié :

  • Film : `"Film : Inception (2010) | Nolan | 148 min"
  • Serie : `"Série : Breaking Bad (2008) | 5 saisons, 65 épisodes"
  • Documentaire : `"Doc : Free Solo (2018) | Alpinisme"

Utilisez base.ToString() pour factoriser. Testez avec un foreach sur une liste mixte.

// TODO : overridez ToString() dans chaque classe

var medias = new List<Media>
{
    new Film("Inception", 2010, "Nolan", 148),
    new Serie("Breaking Bad", 2008, 5, 13),
    new Documentaire("Free Solo", 2018, "Alpinisme", false)
};

foreach (var m in medias)
    Console.WriteLine(m);
Solution
public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }
    public override string ToString() => $"{Titre} ({Annee})";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public override string ToString() => $"Film : {base.ToString()} | {Realisateur} | {DureeMinutes} min";
}

public class Serie : Media
{
    public int Saisons { get; set; }
    public int EpisodesParSaison { get; set; }
    public int NombreTotalEpisodes => Saisons * EpisodesParSaison;
    public override string ToString() => $"Série : {base.ToString()} | {Saisons} saisons, {NombreTotalEpisodes} épisodes";
}

public class Documentaire : Media
{
    public string Sujet { get; set; }
    public override string ToString() => $"Doc : {base.ToString()} | {Sujet}";
}

Exercice 4 : new vs override

Démontrez la différence entre new et override :

  1. Créez Media.GetDescription() (non virtuelle) retournant "Média générique"
  2. Dans Film, masquez-la avec new string GetDescription() retournant "Description du film"
  3. Dans Serie, créez une méthode virtual string GetTag() dans Media et overridez-la
  4. Montrez que via une variable Media, GetDescription() appelle la version de base mais GetTag() appelle la version dérivée
// Montrez la différence entre new et override

Media m1 = new Film();
Console.WriteLine(m1.GetDescription()); // ???
Console.WriteLine(m1.GetTag());          // ???

Film f1 = new Film();
Console.WriteLine(f1.GetDescription()); // ???
Solution
public class Media
{
    public string GetDescription() => "Média générique";
    public virtual string GetTag() => "[MEDIA]";
}

public class Film : Media
{
    public new string GetDescription() => "Description du film";
    public override string GetTag() => "[FILM]";
}

Media m = new Film();
Console.WriteLine(m.GetDescription()); // Média générique (new = compile-time)
Console.WriteLine(m.GetTag());          // [FILM] (override = runtime)

Film f = new Film();
Console.WriteLine(f.GetDescription()); // Description du film

Solution
public abstract class Notification
{
    public string Destinataire { get; set; }
    protected Notification(string destinataire) => Destinataire = destinataire;
    public abstract void Envoyer();
}

public class NotificationEmail : Notification
{
    public string Sujet { get; set; }
    public NotificationEmail(string email, string sujet) : base(email) => Sujet = sujet;
    public override void Envoyer() => Console.WriteLine($"📧 Email à {Destinataire} : {Sujet}");
}

public class NotificationSms : Notification
{
    public string NumeroTelephone { get => Destinataire; }
    public NotificationSms(string numero, string message) : base(numero) { }
    public override void Envoyer() => Console.WriteLine($"📱 SMS au {NumeroTelephone} : Film disponible");
}

public class NotificationPush : Notification
{
    public string Appareil { get; set; }
    public NotificationPush(string appareil, string msg) : base(appareil) => Appareil = appareil;
    public override void Envoyer() => Console.WriteLine($"🔔 Push sur {Appareil} : {Destinataire}");
}
