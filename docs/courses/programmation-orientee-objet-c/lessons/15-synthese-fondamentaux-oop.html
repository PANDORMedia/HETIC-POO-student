L'évolution de Film

// Matin : classes simples
public string Titre;
public int Annee;

// Maintenant : encapsulation complète
public string Titre { get; set; }
private int _annee;
public int Annee
{
    get => _annee;
    set
    {
        if (value < 1888) throw new ArgumentException();
        _annee = value;
    }
}
public int Age => DateTime.Now.Year - _annee;
