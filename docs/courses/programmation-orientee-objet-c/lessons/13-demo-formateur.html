Démo formateur : Classes, Constructeurs & Encapsulation

Ce document est le script de démonstration pour le formateur. Suis les étapes dans l'ordre, en expliquant chaque concept au fur et à mesure.

Durée : 25 minutes

Objectif : Montrer en direct l'évolution d'une classe Film depuis sa création jusqu'à sa forme encapsulée.

Étape 1 : Créer la classe Film (5 min)

Commence par créer un nouveau projet et une classe minimale. Explique chaque élément pendant que tu tapes.

// Créer le projet dans le terminal :
// mkdir DemoFilms && cd DemoFilms && dotnet new console

// Film.cs
public class Film
{
    public string titre;
    public string realisateur;
    public int annee;
    public string genre;

    public void Afficher()
    {
        Console.WriteLine($"{titre} ({annee}) - {realisateur} [{genre}]");
    }
}

Étape 2 : Ajouter des constructeurs (5 min)

Montre le problème du constructeur par défaut, puis ajoute un constructeur paramétré et la surcharge.

// Ajouter à Film.cs
public Film(string titre, string realisateur, int annee, string genre)
{
    this.titre = titre;
    this.realisateur = realisateur;
    this.annee = annee;
    this.genre = genre;
}

public Film(string titre, string realisateur, int annee)
    : this(titre, realisateur, annee, "Non défini")
{
}

// Program.cs
Film f1 = new Film("Inception", "Nolan", 2010, "Sci-Fi");
Film f2 = new Film("Parasite", "Bong", 2019);
Film f3 = new Film("Dune", "Villeneuve", 2021, "Sci-Fi");

f1.Afficher();
f2.Afficher();
f3.Afficher();

Étape 3 : Encapsuler avec des propriétés (7 min)

Maintenant, transforme les champs publics en propriétés encapsulées. Montre la validation dans le setter.

// Film.cs - version encapsulée complète
public class Film
{
    private int _annee;

    public string Titre { get; set; }
    public string Realisateur { get; set; }
    public string Genre { get; set; }

    public int Annee
    {
        get => _annee;
        set
        {
            if (value < 1888 || value > DateTime.Now.Year + 5)
                throw new ArgumentOutOfRangeException(
                    nameof(Annee), $"Annee invalide : {value}");
            _annee = value;
        }
    }

    public int Age => DateTime.Now.Year - Annee;

    public Film(string titre, string realisateur, int annee, string genre = "Non défini")
    {
        Titre = titre;
        Realisateur = realisateur;
        Annee = annee;    // Passe par le setter avec validation !
        Genre = genre;
    }

    public void Afficher()
    {
        Console.WriteLine($"{Titre} ({Annee}) - {Realisateur} [{Genre}] - {Age} ans");
    }
}

Étape 4 : Créer une FilmCollection (5 min)

Montre comment encapsuler une collection avec une liste privée et des méthodes d'accès.

using System.Collections.Generic;

public class FilmCollection
{
    private List<Film> films = new();

    public int Count => films.Count;

    public void Ajouter(Film f) => films.Add(f);

    public void Supprimer(string titre) =>
        films.RemoveAll(f => f.Titre == titre);

    public List<Film> RechercherParGenre(string genre)
    {
        List<Film> resultats = new();
        foreach (Film f in films)
        {
            if (f.Genre == genre) resultats.Add(f);
        }
        return resultats;
    }

    public void AfficherTous()
    {
        Console.WriteLine($"Collection : {Count} films");
        foreach (Film f in films) f.Afficher();
    }
}

Étape 5 : Démo complète dans Program.cs (3 min)

Assemble le tout et montre le résultat final.

// Program.cs - Démo complète
FilmCollection collection = new FilmCollection();

collection.Ajouter(new Film("Inception", "Nolan", 2010, "Sci-Fi"));
collection.Ajouter(new Film("Parasite", "Bong", 2019, "Thriller"));
collection.Ajouter(new Film("Dune", "Villeneuve", 2021, "Sci-Fi"));
collection.Ajouter(new Film("Le Grand Bleu", "Besson", 1988, "Drame"));

collection.AfficherTous();

Console.WriteLine();
Console.WriteLine("Films Sci-Fi :");
foreach (Film f in collection.RechercherParGenre("Sci-Fi"))
    f.Afficher();

collection.Supprimer("Parasite");
Console.WriteLine($"Après suppression : {collection.Count} films");

// Test validation

{
    Film invalide = new Film("Test", "Test", 1800);

{
