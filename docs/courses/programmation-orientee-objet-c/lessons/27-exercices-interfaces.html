Exercice 1 : Interface IStreamable

Solution
public interface IStreamable
{
    string Plateforme { get; set; }
    bool EstDisponibleEnStreaming();
    string GetLienStreaming();
}

public class Film : Media, IStreamable
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public string Plateforme { get; set; }

    public bool EstDisponibleEnStreaming() => !string.IsNullOrEmpty(Plateforme);
    public string GetLienStreaming() => $"https://{Plateforme?.ToLower()}.com/play/{Titre?.Replace(' ', '-')}";
}

Exercice 2 : INotable et IExportable

  1. Créez INotable avec Note (double), AttribuerNote(double) et GetAvis() (string)
  2. Créez IExportable avec ExporterJson() et ExporterCsv() (string)
  3. Faites implémenter les deux interfaces par Film
  4. AttribuerNote doit clamper entre 0 et 10
  5. GetAvis retourne "Recommandé" si >= 7, "Moyen" si >= 5, sinon "À éviter"

{
    // TODO
}

public interface IExportable
{
    // TODO
}

public class Film : Media, INotable, IExportable
{
    // TODO
}
Solution
public interface INotable
{
    double Note { get; set; }
    void AttribuerNote(double note);
    string GetAvis();
}

public interface IExportable
{
    string ExporterJson();
    string ExporterCsv();
}

public class Film : Media, INotable, IExportable
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }

    // INotable
    public double Note { get; set; }
    public void AttribuerNote(double note) => Note = Math.Clamp(note, 0, 10);
    public string GetAvis() => Note >= 7 ? "Recommandé" : Note >= 5 ? "Moyen" : "À éviter";

    // IExportable
    public string ExporterJson() => $"{{\"titre\":\"{Titre}\",\"note\":{Note}}}";
    public string ExporterCsv() => $"{Titre},{Annee},{Note}";
}

Exercice 4 : Découplage avec interface

Refactorez ce code couplé en utilisant une interface :

  2. Montrez qu'il peut maintenant exporter n'importe quel type qui implémente IExportable
  3. Créez une classe SerieExportable : Serie, IExportable pour le démontrer
public class ServiceFilm
{
    // TODO : refactorer pour utiliser IExportable
}
Solution
public class ServiceExport
{
    public void Afficher(IExportable exportable)
    {
        Console.WriteLine(exportable.ExporterCsv());
    }

    public void AfficherTout(IEnumerable<IExportable> exportables)
    {
        foreach (var e in exportables)
            Console.WriteLine(e.ExporterCsv());
    }
}

// Le service accepte n'importe quel IExportable
var service = new ServiceExport();
service.Afficher(new Film { Titre = "Dune" });
service.Afficher(new SerieExportable { Titre = "Dark" });

Exercice 5 (Challenge) : Tri par interface

Créez une interface IComparable<T> (utilisez celle du framework) :

  1. Faites implémenter IComparable<Media> dans Media pour trier par année puis titre
  2. Créez une List<Media> non triée et montrez le tri avec Sort()
  3. Créez un comparateur alternatif ComparateurNote : IComparer<INotable> qui trie par note décroissante
  4. Testez les deux tris
public abstract class Media : IComparable<Media>
{
    // TODO : implémentez CompareTo
}
Solution
public abstract class Media : IComparable<Media>
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public int CompareTo(Media? other)
    {
        if (other is null) return 1;
        int cmp = Annee.CompareTo(other.Annee);
        return cmp != 0 ? cmp : string.Compare(Titre, other.Titre, StringComparison.Ordinal);
    }
}

public class ComparateurNote : IComparer<INotable>
{
    public int Compare(INotable? x, INotable? y)
    {
        if (x is null) return 1;
        if (y is null) return -1;
        return y.Note.CompareTo(x.Note); // Décroissant
    }
}

// Tri par année/titre
var medias = new List<Media> { /* ... */ };
medias.Sort();

// Tri par note
var notables = new List<INotable> { /* ... */ };
notables.Sort(new ComparateurNote());
