public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public Media(string titre, int annee)
    {
        Titre = titre;
        Annee = annee;
    }

    public virtual string GetInfo() => $"{Titre} ({Annee})";
    public override string ToString() => $"[{GetInfo()}]";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }

    public Film(string titre, int annee, string realisateur, int duree)
        : base(titre, annee)
    {
        Realisateur = realisateur;
        DureeMinutes = duree;
    }

    public override string ToString()
        => $"[{Titre} ({Annee}) - {Realisateur}, {DureeMinutes} min]";
}

public class Serie : Media
{
    public int Saisons { get; set; }
    public int EpisodesParSaison { get; set; }

    public Serie(string titre, int annee, int saisons, int episodes)
        : base(titre, annee)
    {
        Saisons = saisons;
        EpisodesParSaison = episodes;
    }

    public int NombreTotalEpisodes => Saisons * EpisodesParSaison;
    public override string ToString() => $"[{Titre} ({Annee}) - {NombreTotalEpisodes} épisodes]";
}

public class Documentaire : Media
{
    public string Sujet { get; set; }
    public bool EstBiographie { get; set; }

    public Documentaire(string titre, int annee, string sujet, bool estBio)
        : base(titre, annee)
    {
        Sujet = sujet;
        EstBiographie = estBio;
    }

    public override string ToString()
        => EstBiographie
            ? $"[{Titre} ({Annee}) - {Sujet} [Bio]]"
            : $"[{Titre} ({Annee}) - {Sujet}]";
}

static void AfficherDetails(Media media)
{
    Console.WriteLine($"Type : {media.GetType().Name}");
    Console.WriteLine($"Info : {media.GetInfo()}");

    if (media is Film film)
        Console.WriteLine($"  Réalisateur : {film.Realisateur}, Durée : {film.DureeMinutes} min");
    else if (media is Serie serie)
        Console.WriteLine($"  Saisons : {serie.Saisons}, Total épisodes : {serie.NombreTotalEpisodes}");
    else if (media is Documentaire doc)
        Console.WriteLine($"  Sujet : {doc.Sujet}, Bio : {doc.EstBiographie}");

    Console.WriteLine();
}

public class Media
{
    private static int _compteur = 0;
    protected int _id;
    protected int _noteInterne;

    public string Titre { get; set; }
    public int Annee { get; set; }

    public Media(string titre, int annee)
    {
        _id = ++_compteur;
        Titre = titre;
        Annee = annee;
        _noteInterne = -1;
    }
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }

    public Film(string titre, int annee, string realisateur, int duree)
        : base(titre, annee)
    {
        Realisateur = realisateur;
        DureeMinutes = duree;
    }

    public void SetNote(int note)
    {
        _noteInterne = note; // Accessible car protected
    }
}

public sealed class FilmStreaming : Film
{
    public string Plateforme { get; set; }

    public FilmStreaming(string titre, int annee, string realisateur, int duree, string plateforme)
        : base(titre, annee, realisateur, duree)
    {
        Plateforme = plateforme;
    }

    public override string ToString()
        => $"[{Titre} ({Annee}) - {Realisateur} via {Plateforme}]";
}

Exercice 5 (Challenge) : Catalogue polymorphe

Créez une classe Catalogue qui :

  1. Contient une List<Media>
  2. A une méthode Ajouter(Media media) qui empêche les doublons de titre
  3. A une méthode RechercherParType<T>() qui retourne tous les médias d'un type donné
  4. A une méthode GetStatistiques() retournant le nombre de films, séries et documentaires

cat.Ajouter(new Film("Dune", 2021, "Villeneuve", 155));
cat.Ajouter(new Serie("Severance", 2022, 2, 10));
cat.Ajouter(new Film("Parasite", 2019, "Bong", 132));

Console.WriteLine(cat.GetStatistiques());
// Films : 2, Séries : 1, Documentaires : 0

    Console.WriteLine(film);

Solution
public class Catalogue
{
    private List<Media> _medias = new();

    public void Ajouter(Media media)
    {
        if (_medias.Any(m => m.Titre == media.Titre))
            throw new InvalidOperationException($"Un média nommé '{media.Titre}' existe déjà.");
        _medias.Add(media);
    }

    public IEnumerable<T> RechercherParType<T>() where T : Media
        => _medias.OfType<T>();

    public string GetStatistiques()
    {
        int films = _medias.Count(m => m is Film);
        int series = _medias.Count(m => m is Serie);
        int docs = _medias.Count(m => m is Documentaire);
        return $"Films : {films}, Séries : {series}, Documentaires : {docs}";
    }
}
