Film : Media, la syntaxe de l'héritage

  • L'opérateur : établit l'héritage
  • Film obtient Titre, Annee et GetInfo()
public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public string GetInfo() => $"{Titre} ({Annee})";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
}

var film = new Film
{
    Titre = "Inception",
    Annee = 2010,
    Realisateur = "Christopher Nolan",
    DureeMinutes = 148
};
Console.WriteLine(film.GetInfo()); // Inception (2010)

Slide 6 / 12

base() appelle le constructeur parent

  • base() transmet les paramètres au parent
  • Le constructeur parent s'exécute en premier
public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public Media(string titre, int annee)
    {
        Titre = titre;
        Annee = annee;
    }
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }

    public Film(string titre, int annee, string realisateur, int duree)
        : base(titre, annee)
    {
        Realisateur = realisateur;
        DureeMinutes = duree;
    }
}

Upcasting, downcasting et pattern matching

  • Upcasting : toujours sûr, implicite
  • Downcasting : explicite, peut échouer
  • is : pattern matching sûr et concis
Film film = new Film("Parasite", 2019, "Bong Joon-ho", 132);
Media media = film; // Upcasting implicite (sûr)
Console.WriteLine(media.Titre); // Parasite

// Downcasting explicite
Media m2 = new Film("Amélie", 2001, "Jeunet", 122);
Film f2 = (Film)m2;

// Pattern matching avec is
if (m2 is Film f)
    Console.WriteLine($"Film de {f.Realisateur}");

// Opérateur as
Film f3 = m2 as Film;

sealed : verrouiller l'héritage

  • Aucune classe ne peut hériter de FilmStreaming
  • Cas d'usage : protéger un comportement critique
public sealed class FilmStreaming : Film
{
    public string Plateforme { get; set; }

    public FilmStreaming(string titre, int annee,
        string realisateur, int duree, string plateforme)
        : base(titre, annee, realisateur, duree)
    {
        Plateforme = plateforme;
    }
}
