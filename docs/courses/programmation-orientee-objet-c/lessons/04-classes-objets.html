class Film
{
    public string Titre;
    public int Annee;
    public string Genre;

    public void Afficher()
    {
        Console.WriteLine($"{Titre} ({Annee}) - {Genre}");
    }
}

Créer des objets avec new

Film film1 = new Film();

film1.Afficher();
// Inception (2010) - Sci-Fi

Film film2 = new Film();
film2.Titre = "Matrix";
film2.Annee = 1999;
film2.Afficher();
// Matrix (1999) - Sci-Fi
