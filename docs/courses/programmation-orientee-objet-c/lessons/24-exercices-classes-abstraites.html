Exercice 1 : Media abstrait

Transformez Media en classe abstraite :

  1. Déclarez Media comme abstract
  2. Ajoutez une méthode abstraite string GetTypeMedia()
  3. Ajoutez une méthode abstraite decimal CalculerPrix()
  4. Implémentez ces méthodes dans Film, Serie et Documentaire
  5. Vérifiez que new Media() provoque une erreur de compilation
public abstract class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }
    // TODO : méthodes abstraites
}

public class Film : Media
{
    // TODO
}
Solution
public abstract class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }
    protected Media(string titre, int annee) { Titre = titre; Annee = annee; }
    public abstract string GetTypeMedia();
    public abstract decimal CalculerPrix();
    public override string ToString() => $"{GetTypeMedia()} : {Titre} ({Annee}) - {CalculerPrix():C}";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public Film(string t, int a, string r, int d) : base(t, a) { Realisateur = r; DureeMinutes = d; }
    public override string GetTypeMedia() => "Film";
    public override decimal CalculerPrix() => DureeMinutes > 120 ? 14.99m : 9.99m;
}

public class Serie : Media
{
    public int Saisons { get; set; }
    public Serie(string t, int a, int s) : base(t, a) => Saisons = s;
    public override string GetTypeMedia() => "Série";
    public override decimal CalculerPrix() => Saisons * 7.99m;
}

public class Documentaire : Media
{
    public string Sujet { get; set; }
    public Documentaire(string t, int a, string s) : base(t, a) => Sujet = s;
    public override string GetTypeMedia() => "Documentaire";
    public override decimal CalculerPrix() => 6.99m;
}

Exercice 2 : Template Method GenererFiche

Ajoutez une méthode concrète GenererFiche() dans Media qui appelle les méthodes abstraites :

=== Film ===
Titre : Dune
Année : 2021
Détails : Réalisateur : Villeneuve, Durée : 155 min
Prix : 14,99 EUR
  1. GenererFiche() est concrète dans Media et ne peut pas être override
  2. Elle appelle GetTypeMedia(), GetDetails() et CalculerPrix()
  3. Chaque classe dérivée implémente GetDetails()
public abstract class Media
{
    // TODO : GenererFiche() + méthodes abstraites
}
Solution
public abstract class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public string GenererFiche()
    {
        return $"=== {GetTypeMedia()} ===\n" +
               $"Titre : {Titre}\n" +
               $"Année : {Annee}\n" +
               $"Détails : {GetDetails()}\n" +
               $"Prix : {CalculerPrix():C}";
    }

    protected abstract string GetTypeMedia();
    protected abstract string GetDetails();
    public abstract decimal CalculerPrix();
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    protected override string GetTypeMedia() => "Film";
    protected override string GetDetails() => $"Réalisateur : {Realisateur}, Durée : {DureeMinutes} min";
    public override decimal CalculerPrix() => DureeMinutes > 120 ? 14.99m : 9.99m;
}

Exercice 3 : Comparaison abstract vs virtual

Créez une classe Media qui mélange méthodes abstraites et virtuelles :

  1. abstract string GetCategorie() : chaque dérivée DOIT la fournir
  2. virtual string GetInfo() => "..." : un défaut existe, les dérivées peuvent l'override
  3. Dans Film, overridez les deux
  4. Dans une nouvelle classe CourtMetrage : Media, overridez uniquement GetCategorie() et laissez le défaut de GetInfo()

Montrez que CourtMetrage.GetInfo() utilise le défaut sans erreur.

public abstract class Media
{
    // TODO : mélange abstract + virtual
}

public class CourtMetrage : Media
{
    // TODO : overridez uniquement GetCategorie
}
Solution
public abstract class Media
{
    public string Titre { get; set; }
    public abstract string GetCategorie(); // Obligatoire
    public virtual string GetInfo() => $"{GetCategorie()} : {Titre}"; // Défaut
}

public class Film : Media
{
    public override string GetCategorie() => "Film";
    public override string GetInfo() => $"Film : {Titre} (long métrage)";
}

public class CourtMetrage : Media
{
    public int DureeSecondes { get; set; }
    public override string GetCategorie() => "Court-métrage";
    // GetInfo() utilise le défaut : "Court-métrage : {Titre}"
}

Exercice 4 : Système de rapport

Créez une classe abstraite RapportMedia :

  1. Méthode abstraite string GetEntete()
  2. Méthode abstraite string GetCorps()
  3. Méthode concrète string GenererRapport() qui assemble entête, corps et pied de page
  4. RapportFilm, RapportSerie qui héritent de RapportMedia
  5. Testez avec une List<RapportMedia>
public abstract class RapportMedia
{
    // TODO
}

// TODO : RapportFilm, RapportSerie
Solution
public abstract class RapportMedia
{
    protected abstract string GetEntete();
    protected abstract string GetCorps();

    public string GenererRapport()
    {
        return $"{GetEntete()}\n{GetCorps()}\n--- Fin du rapport ---\n";
    }
}

public class RapportFilm : RapportMedia
{
    private readonly Film _film;
    public RapportFilm(Film film) => _film = film;
    protected override string GetEntete() => $"RAPPORT FILM : {_film.Titre}";
    protected override string GetCorps() => $"Réalisateur : {_film.Realisateur}\nDurée : {_film.DureeMinutes} min";
}

public class RapportSerie : RapportMedia
{
    private readonly Serie _serie;
    public RapportSerie(Serie serie) => _serie = serie;
    protected override string GetEntete() => $"RAPPORT SÉRIE : {_serie.Titre}";
    protected override string GetCorps() => $"Saisons : {_serie.Saisons}";
}

Exercice 5 (Challenge) : Hiérarchie complète

Construisez la hiérarchie finale du gestionnaire de films :

  1. abstract class Media avec Id auto-incrémenté, Titre, Annee, constructeur protected
  2. Méthodes abstraites : GetTypeMedia(), CalculerPrix(), GetDetails()
  3. Méthode concrète : GenererFiche() (template method)
  4. Film, Serie, Documentaire complètement implémentées
  5. Un programme principal qui crée un catalogue et affiche toutes les fiches
public abstract class Media
{
    // TODO : hiérarchie complète
}
Solution
public abstract class Media
{
    private static int _compteur = 0;
    public int Id { get; }
    public string Titre { get; set; }
    public int Annee { get; set; }

    protected Media(string titre, int annee)
    {
        Id = ++_compteur;
        Titre = titre;
        Annee = annee;
    }

    protected abstract string GetTypeMedia();
    protected abstract string GetDetails();
    public abstract decimal CalculerPrix();

    public string GenererFiche()
        => $"#{Id} [{GetTypeMedia()}] {Titre} ({Annee})\n  {GetDetails()}\n  Prix : {CalculerPrix():C}";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public string Genre { get; set; }
    public Film(string t, int a, string r, int d, string g) : base(t, a)
        { Realisateur = r; DureeMinutes = d; Genre = g; }
    protected override string GetTypeMedia() => "Film";
    protected override string GetDetails() => $"{Realisateur} | {DureeMinutes} min | {Genre}";
    public override decimal CalculerPrix() => DureeMinutes > 120 ? 14.99m : 9.99m;
}

public class Serie : Media
{
    public int Saisons { get; set; }
    public Serie(string t, int a, int s) : base(t, a) => Saisons = s;
    protected override string GetTypeMedia() => "Série";
    protected override string GetDetails() => $"{Saisons} saison(s)";
    public override decimal CalculerPrix() => Saisons * 7.99m;
}

public class Documentaire : Media
{
    public string Sujet { get; set; }
    public Documentaire(string t, int a, string s) : base(t, a) => Sujet = s;
    protected override string GetTypeMedia() => "Doc";
    protected override string GetDetails() => $"Sujet : {Sujet}";
    public override decimal CalculerPrix() => 6.99m;
}
