Polymorphisme

virtual, override et dispatch dynamique

  • Un seul appel, des comportements différents
  • Le pilier qui rend l'héritage puissant

virtual + override : le duo essentiel

  • virtual : autorise la redéfinition dans les classes dérivées
  • override : remplace effectivement le comportement
  • Sans virtual, pas de polymorphisme possible
Slide 3 / 13

Variable Media, objet Film : que se passe-t-il ?

  • Variable de type Media, objet de type Film
  • Quelle version de GetInfo() est appelée ?
  • La réponse dépend du runtime, pas du compile-time
Slide 4 / 13

GetInfo() polymorphe

public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }

    public virtual string GetInfo() => $"{Titre} ({Annee})";
}

public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }

    public override string GetInfo()
        => $"{Titre} ({Annee}) - {Realisateur}, {DureeMinutes} min";
}

public class Serie : Media
{
    public int Saisons { get; set; }

    public override string GetInfo()
        => $"{Titre} ({Annee}) - {Saisons} saison(s)";
}

List<Media> : un foreach, quatre comportements

List<Media> mediatheque = new()
{
    new Film { Titre = "Inception", Annee = 2010,
               Realisateur = "Nolan", DureeMinutes = 148 },
    new Serie { Titre = "Breaking Bad", Annee = 2008,
                Saisons = 5 },
    new Film { Titre = "Parasite", Annee = 2019,
               Realisateur = "Bong Joon-ho", DureeMinutes = 132 },
    new Serie { Titre = "Dark", Annee = 2017, Saisons = 3 }
};

foreach (var media in mediatheque)
    Console.WriteLine(media.GetInfo());
// Chaque GetInfo() appelle la bonne version !
Slide 7 / 13

ToString() : redéfinir l'affichage par défaut

  • Console.WriteLine appelle ToString() automatiquement
  • Utile pour le débogage et l'affichage
  • Composez avec base.ToString() pour factoriser
public class Film : Media
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public string Genre { get; set; }

    public override string ToString()
        => $"{base.ToString()} | {Realisateur} | {DureeMinutes} min | {Genre}";
}

var film = new Film { Titre = "Blade Runner 2049", Annee = 2017,
    Realisateur = "Villeneuve", DureeMinutes = 164, Genre = "Sci-Fi" };
Console.WriteLine(film);
// Blade Runner 2049 (2017) | Villeneuve | 164 min | Sci-Fi
Slide 8 / 13

base. : composer avec le parent

  • base.GetInfo() réutilise le comportement parent
  • Évite de dupliquer la logique commune
  • Utile quand la dérivée enrichit sans remplacer
public override string GetInfo()
{
    return base.GetInfo() + $" - {Realisateur}, {DureeMinutes} min";
}
Slide 9 / 13

Prix polymorphe : chaque type, sa logique

public class Media
{
    public string Titre { get; set; }
    public int Annee { get; set; }
    public virtual decimal CalculerPrix() => 9.99m;
}

public class Film : Media
{
    public int DureeMinutes { get; set; }
    public override decimal CalculerPrix()
        => DureeMinutes > 120 ? 14.99m : 9.99m;
}

public class Serie : Media
{
    public int Saisons { get; set; }
    public override decimal CalculerPrix() => Saisons * 7.99m;
}

List<Media> catalogue = new()
{
    new Film { Titre = "Dune", DureeMinutes = 155 },
    new Serie { Titre = "Severance", Saisons = 2 }
};

decimal total = 0;
foreach (var m in catalogue)
{
    Console.WriteLine($"{m.Titre} : {m.CalculerPrix():C}");
    total += m.CalculerPrix();
}
Console.WriteLine($"Total : {total:C}");

new vs override : ne pas confondre

  • override : dispatch dynamique selon le type de l'objet
