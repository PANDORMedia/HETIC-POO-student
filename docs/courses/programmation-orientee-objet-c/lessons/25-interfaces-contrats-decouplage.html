Définir et implémenter une interface

public interface IStreamable
{
    string Plateforme { get; set; }
    bool EstDisponibleEnStreaming();
    string GetLienStreaming();
}

public class Film : Media, IStreamable
{
    public string Realisateur { get; set; }
    public int DureeMinutes { get; set; }
    public string Plateforme { get; set; }

    public bool EstDisponibleEnStreaming()
        => !string.IsNullOrEmpty(Plateforme);
    public string GetLienStreaming()
        => $"https://{Plateforme?.ToLower()}.com/play/{Titre}";
}
Slide 3 / 11

Implémentation multiple : IStreamable + INotable + IExportable

public interface INotable
{
    double Note { get; set; }
    void AttribuerNote(double note);
    string GetAvis();
}

public interface IExportable
{
    string ExporterJson();
    string ExporterCsv();
}

public class Film : Media, IStreamable, INotable, IExportable
{
    // IStreamable
    public string Plateforme { get; set; }
    public bool EstDisponibleEnStreaming() => !string.IsNullOrEmpty(Plateforme);
    public string GetLienStreaming() => $"https://{Plateforme?.ToLower()}.com/{Titre}";

    // INotable
    public double Note { get; set; }
    public void AttribuerNote(double note) => Note = Math.Clamp(note, 0, 10);
    public string GetAvis() => Note >= 7 ? "Recommandé" : Note >= 5 ? "Moyen" : "À éviter";

    // IExportable
    public string ExporterJson() => $"{{\"titre\":\"{Titre}\",\"note\":{Note}}}";
    public string ExporterCsv() => $"{Titre},{Annee},{Note},{Realisateur}";
}

Polymorphisme par interface

Film film = new Film
{
    Titre = "Dune",
    Annee = 2021,
    Realisateur = "Villeneuve",
    DureeMinutes = 155,
    Plateforme = "Netflix"
};

// Polymorphisme par interface
INotable notable = film;
notable.AttribuerNote(8.5);
Console.WriteLine(notable.GetAvis()); // Recommandé

IStreamable streamable = film;
Console.WriteLine(streamable.EstDisponibleEnStreaming()); // True

IExportable exportable = film;
Console.WriteLine(exportable.ExporterCsv()); // Dune,2021,8.5,Villeneuve

Slide 7 / 11

Découplage avec les interfaces

  • Le service accepte n'importe quel IExportable
  • Pas besoin de connaître la classe concrète
  • Testabilité et flexibilité améliorées
// Mauvais : couplage fort
public class ServiceFilm
{
    public void Afficher(Film film)
        => Console.WriteLine(film.ExporterCsv());
}

// Bon : couplage faible
public class ServiceExport
{
    public void Afficher(IExportable exportable)
        => Console.WriteLine(exportable.ExporterCsv());
}

Interfaces en résumé

  • Interface = contrat pur, sans état
  • Implémentation multiple : une classe, plusieurs contrats
  • Polymorphisme par interface : manipuler par capacité
  • Découplage : dépendre des abstractions, pas des concrétions
  • Prochaine étape : SOLID et composition
